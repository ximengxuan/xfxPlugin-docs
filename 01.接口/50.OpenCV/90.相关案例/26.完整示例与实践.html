3. 多边形检测与轮廓分析 ​

lua
local opcvOps = UtilsMain.opencvOps()
local bmp = LuaEngine.snapShot(0, 0, 0, 0)
local srcMat = opcvOps.bitmapToMat(bmp)
local grayMat = opcvOps.gray(srcMat)
local binaryMat = opcvOps.threshold(grayMat, 127, 255, 0)

-- 检测三角形（edgeCount=3 等于，minArea=100，返回完整 JSON）
local result = opcvOps.findPolygons(binaryMat, 3, 0, 0, 2, 100, -1, -1, -1)
local obj = jsonLib.decode(result)
if obj.code == 0 then
    for i, item in ipairs(obj.data) do
        print("三角形 center=(" .. item.centerX .. "," .. item.centerY .. ") edgeCount=" .. item.edgeCount)
    end
end

-- 获取全部轮廓的外接矩形与面积
local rects = opcvOps.getContoursRect(binaryMat, 0, 2, "")
local areas = opcvOps.getContoursArea(binaryMat, 0, 2, "")
local lengths = opcvOps.getContoursLength(binaryMat, 0, 2, "")
local moments = opcvOps.getContoursMoments(binaryMat, 0, 2, "")
print("矩形:", rects, "面积:", areas, "长度:", lengths, "重心:", moments)

opcvOps.clear()

6. 裁剪、水印、圆角、压缩与保存相册 ​

lua
local UtilsMain = loader.loadClass('com.xfx.plugin.UtilsMain')
UtilsMain.init(LuaEngine.getContext())

local opcvOps = UtilsMain.opencvOps()
local bmp = LuaEngine.snapShot(0, 0, 0, 0)
local srcMat = opcvOps.bitmapToMat(bmp)

local info = jsonLib.decode(opcvOps.getInfo(srcMat))
print("原图:", info.width, info.height, "通道:", info.channels)

-- clip 使用 left/top/width/height，更适合脚本侧 ROI 写法
local roi = opcvOps.clip(srcMat, 100, 160, 360, 240)

-- 文字水印
local textMarked = opcvOps.addTextWatermark(roi, "xfxPlugin", 32, 0xAAFFFFFF, 16.0, 48.0)

-- 图片水印
local logo = opcvOps.readResMat("/sdcard/Download/logo.png")
local imageMarked = opcvOps.addImageWatermark(textMarked, logo, 16, 16, 180)

-- 圆角与压缩
local rounded = opcvOps.round(imageMarked, 24)
local compressed = opcvOps.compress(rounded, 0.6, 0.6, 80, 1)

opcvOps.save(compressed, "/sdcard/Download/opencv_clip_watermark.jpg", "JPG", 85)
local albumPath = opcvOps.saveToAlbum(compressed, "PNG", 100)
print("保存到相册:", albumPath)

opcvOps.clear()

OOP 例子 ​

基础滤镜 ​

演示了如何根据 CPU 架构，加载指定的 ABI 插件包（体积比全架构小很多）。

lua
-- 懒人精灵高级版
import('com.nx.assist.lua.LuaEngine')

-- 获取 CPU 架构名称
local function getCpuArchName()
    local arch = getCpuArch()
    if arch == 0 then
        return 'x86'
    elseif arch == 1 then
        return 'armeabi-v7a'
    elseif arch == 2 then
        return 'arm64-v8a'
    elseif arch == 3 then
        return 'x86_64'
    end
end

-- 加载 XfxPlugin.lua 类
local xfxModule = require('lib/XfxPlugin')

-- 创建 XFX 对象实例
local XFX = xfxModule:new({
    apkName = 'xfxPlugin-release' .. '-' .. getCpuArchName() ..'.apk',  -- 加载指定CPU架构的 APK 文件名
    context = LuaEngine.getContext(), -- 插件的全局上下文
})

---------------------------------------------------------------------------------------------------------------------

-- 引入图像识别类
local opcvOps = XFX:getOps('opencvOps'); assert(opcvOps)

-- 懒人精灵高级版截图
local bmp = LuaEngine.snapShot(0, 0, 0, 0)
local srcMat = opcvOps.bitmapToMat(bmp)   -- 统一转换入口

local grayMat = opcvOps.gray(srcMat)                                      -- 灰度，返回Mat句柄
local thresholdMat = opcvOps.threshold(grayMat, 127, 255, 0)                -- BINARY 二值化
local dbThresholdMat = opcvOps.doubleThreshold(grayMat, 100, 200, 255)      -- 双阈值二值化（保留 100~200 亮度区间的像素）
local adaptiveMat = opcvOps.adaptiveThreshold(grayMat, 255, 1, 0, 15, 10)   -- 高斯加权自适应二值化，块大小 15，常数 10

-- 保存
local path = '/sdcard/Download/opencvTest/'
print('保存灰度图',opcvOps.save(grayMat, path .. "gray.png"))
print('保存二值化图',opcvOps.save(thresholdMat, path .. "threshold.png"))
print('保存双阈值图',opcvOps.save(dbThresholdMat, path .. "dbThreshold.png"))
print('保存自适应图',opcvOps.save(adaptiveMat, path .. "adaptive.png"))

-- 回收方式一：逐个回收
print('回收原Mat图',opcvOps.recycle(srcMat))
print('回收灰度图',opcvOps.recycle(grayMat))
print('回收二值化图',opcvOps.recycle(thresholdMat))
print('回收双阈值图',opcvOps.recycle(dbThresholdMat))
print('回收自适应图',opcvOps.recycle(adaptiveMat))

-- 回收方式二：清空全部Mat句柄
opcvOps.clear()

Steam 账号切换：检测矩形头像框 ​

lua
import('com.nx.assist.lua.LuaEngine')

local xfxModule = require('lib/XfxPlugin')
local XFX = xfxModule:new({
    apkName = 'xfxPlugin-release.apk',
    context = LuaEngine.getContext(),
})

local opcvOps = XFX:getOps('opencvOps'); assert(opcvOps)

local imagePath = '/sdcard/Download/opencvTest/测试大图/截图_1785809395573.png'

local function rectJson(list)
  local arr = {}
  for i = 1, #list do
    local r = list[i]
    arr[#arr + 1] = string.format('[%d,%d,%d,%d]', r.left, r.top, r.right, r.bottom)
  end
  return '[' .. table.concat(arr, ',') .. ']'
end

-- 多边形检测
local matId = opcvOps.readResMat(imagePath)
local result = opcvOps.findPolygonsEx(matId, 4, 0, 0, 2, 8000, -1, -1, 0.5)
printEx(result)

local obj = jsonLib.decode(result)
if obj.code == 0 then
  local avatarRects = {}
  for i, item in ipairs(obj.data) do
    if item.left < 220 and item.top > 1600 and item.edgeCount == 4 then
      printEx('矩形 center=(' .. item.centerX .. ',' .. item.centerY .. ') area=' .. item.area)
      avatarRects[#avatarRects + 1] = item
    end
  end

  if #avatarRects > 0 then
    local boxed = opcvOps.drawRects(matId, rectJson(avatarRects), '#53EAFD', 4, false, true)
    opcvOps.save(boxed, '/sdcard/Download/opencvTest/矩形检测结果.png')
  end
  opcvOps.clear()
else
  printEx(obj.message)
end

性能优化策略 ​

1. 链式复用 Mat 句柄 ​

处理链路全程用 Mat 句柄传递，避免每次重新转 Bitmap：

lua
local srcMat = opcvOps.bitmapToMat(bmp)
local grayMat = opcvOps.gray(srcMat)
local resultMat = opcvOps.threshold(grayMat, 127, 255, 0)
opcvOps.recycle(srcMat)
opcvOps.recycle(grayMat)
opcvOps.recycle(resultMat)

2. 及时释放 ​

  • 每个 Mat 句柄用完即 recycle
  • 脚本结束时调用 clear() 释放全部
  • 循环处理时每轮产生的 Mat 及时回收

3. 算法选择 ​

场景推荐
光照均匀的固定阈值分割threshold（BINARY）
光照不均、背景渐变adaptiveThreshold
提取固定亮度区间doubleThreshold
前景亮度不确定threshold（OTSU，type=8）
颜色目标提取getHSV + inRange + copyTo
凸包形状识别（二值图）findPolygons
凸包形状识别（任意图、矩形检测）findPolygonsEx

错误处理机制 ​

图像接口（返回 Mat 句柄）通过判空处理异常：

lua
local resultMat = opcvOps.gray(srcMat)
if resultMat ~= nil and resultMat ~= "" then
    -- 成功，resultMat 为灰度句柄
else
    -- 失败，可能是句柄不存在、native 库未加载、参数非法
end

防御性编程建议：

  1. 每次调用后判断返回值是否为 "" 或 nil
  2. adaptiveThreshold 的 blockSize 必须为正奇数，否则返回 ""
  3. doubleThreshold 的 lowThresh 必须 <= highThresh
  4. 轮廓接口的 contoursIndex 默认传空字符串 "" 表示全部
  5. Mat 句柄使用完毕后务必 recycle 释放
