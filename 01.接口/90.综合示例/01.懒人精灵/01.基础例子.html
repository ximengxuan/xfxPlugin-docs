懒人精灵调用示例（Lua） ​

以下示例基于懒人精灵高级版开发环境，演示如何加载打包好的apk插件并调用各工具方法。 仅使用基础类型与字符串参数，Lua 侧无需处理列表或数组。

综合示例1 ​

lua
import('com.nx.assist.lua.LuaEngine')

-- 1. 加载 APK 与入口类
local loader = LuaEngine.loadApk('xfxPlugin-release.apk'); assert(loader)
local UtilCls = loader.loadClass('com.xfx.plugin.UtilsMain'); assert(UtilCls, '插件入口类未找到')

-- 2. 初始化（传入当前 Context）
local context = LuaEngine.getContext()
UtilCls.init(context)

-- 3. 字符串工具
local strOps = UtilCls.stringOps()
print(strOps.upperFirstLetter('hello'))
print(strOps.reverse('abc123'))
print(strOps.extractUrlsAsJson('官网 https://example.com 与博客 http://blog.example.com'))

-- 4. 文件工具
local fileOps = UtilCls.fileOps()
local base = '/sdcard/xfx_demo/'
local path = base .. 'hello.txt'
fileOps.create(base)
fileOps.write(path, 'Hello Lua!')
print('内容: ' .. tostring(fileOps.readFile2String(path)))
print('大小: ' .. tostring(fileOps.getSize(path)))
fileOps.delete(path)

-- 5. 正则工具
local regOps = UtilCls.regexOps()
print('手机号合法? ' .. tostring(regOps.isMobileSimple('13800138000')))

-- 6. 时间工具
local timeOps = UtilCls.timeOps()
print('当前时间: ' .. timeOps.getNowString())

-- 7. 设备信息
local devOps = UtilCls.deviceOps()
print('设备型号: ' .. devOps.getManufacturer() .. ' / ' .. devOps.getModel())

lua
import('com.nx.assist.lua.LuaEngine')

local loader = LuaEngine.loadApk('xfxPlugin-release.apk'); assert(loader)
local UtilCls = loader.loadClass('com.xfx.plugin.UtilsMain'); assert(UtilCls, '插件入口类未找到')
UtilCls.init(LuaEngine.getContext())

------------------------------------------------------------------------------------------

-- 获取字符串工具
local strOps = UtilCls.stringOps()
print(strOps.upperFirstLetter('test'))
print(strOps.isEmpty('test'))
print(strOps.reverse('test'))

-- 提取字串中的所有網址，返回 JSON 陣列字串
local text = '官方網站 https://www.example.com 以及部落格 http://blog.example.com/page?id=1'
local urlsJson = strOps.extractUrlsAsJson(text)
print(jsonLib.decode(urlsJson))

------------------------------------------------------------------------------------------


-- 获取文件工具
local fileOps = UtilCls.fileOps()

-- 範例路徑
local basePath = '/sdcard/xfx_demo/'
local filePath = basePath .. 'hello.txt'

-- A. 確保目錄／檔案存在
fileOps.create(basePath)
fileOps.create(filePath)

-- B. 寫入文字
local okWrite = fileOps.write(filePath, 'Hello from Lua\nline2')
print('寫入結果: ' .. tostring(okWrite))

-- C. 讀取文字
local content = fileOps.readFile2String(filePath)
print('讀取內容: ' .. tostring(content))

-- D. 取得檔案資訊
print('是否存在: ' .. tostring(fileOps.isExists(filePath)))
print('是否為檔案: ' .. tostring(fileOps.isFile(filePath)))
print('所在資料夾: ' .. tostring(fileOps.getName(filePath)))
print('檔名: ' .. tostring(fileOps.getName(filePath)))
print('不含副檔名: ' .. tostring(fileOps.getName(filePath,false)))
print('副檔名: ' .. tostring(fileOps.getExtension(filePath)))
print('檔案大小: ' .. tostring(fileOps.getSize(filePath)))
print('行數: ' .. tostring(fileOps.getLines(filePath)))
print('MD5: ' .. tostring(fileOps.md5(filePath)))

-- E. 複製／移動
local copyPath = basePath .. 'hello_copy.txt'
fileOps.copy(filePath, copyPath)
print('複製完成? ' .. tostring(fileOps.isExists(copyPath)))

local movePath = basePath .. 'moved/hello.txt'
fileOps.create(basePath .. 'moved/')
fileOps.move(copyPath, movePath)
print('移動完成? ' .. tostring(fileOps.isExists(movePath)))

-- F. 刪除檔案／資料夾
fileOps.delete(filePath)
fileOps.delete(movePath)
fileOps.delete(basePath)
print('刪除後 basePath 是否仍存在: ' .. tostring(fileOps.isDir(basePath)))

------------------------------------------------------------------------------------------

-- 获取正则工具
local regOps = UtilCls.regexOps()
local isMobile = regOps.isMobileSimple('13800138000')
print(isMobile)

------------------------------------------------------------------------------------------

-- 获取时间工具
local timeOps = UtilCls.timeOps()
local now = timeOps.getNowString()
print(now)

------------------------------------------------------------------------------------------

local batteryOps = UtilCls.batteryOps()
print(batteryOps.isIgnoringBatteryOptimizations())

-- 栏操作
local barOps = UtilCls.barOps()
print(barOps.getStatusBarHeight())
print(barOps.getActionBarHeight())
print(barOps.getNavBarHeight())
print(barOps.isSupportNavBar())

-- 拿到亮度工具
local brightnessOps = UtilCls.brightnessOps()
-- 查詢是否開啟自動亮度
local isAuto = brightnessOps.isAutoBrightnessEnabled()
print('自動亮度是否開啟: ' .. tostring(isAuto))

-- 取得當前系統亮度
local current = brightnessOps.getBrightness()
print('當前系統亮度: ' .. tostring(current))

pcall(
    function()
        -- 設定系統亮度為 128(依赖ROOT)
        local ok3 = brightnessOps.setBrightness(128)
        print('設置系統亮度到 128 結果: ' .. tostring(ok3))
    end
)

------------------------------------------------------------------------------------------

-- 设备信息
local deviceOps = UtilCls.deviceOps()

-- A. 判斷 Root / ADB
local rooted = deviceOps.isDeviceRooted()
local adbEnabled = deviceOps.isAdbEnabled()
print('是否 Root: ' .. tostring(rooted))
print('ADB 是否開啟: ' .. tostring(adbEnabled))

-- B. Android 版本資訊
local sdkName = deviceOps.getSDKVersionName()
local sdkCode = deviceOps.getSDKVersionCode()
print('SDK 名稱: ' .. tostring(sdkName))
print('SDK Code: ' .. tostring(sdkCode))

-- C. 基本識別資訊
local androidId = deviceOps.getAndroidID()
local macAddr = deviceOps.getMacAddress()
local manufacturer = deviceOps.getManufacturer()
local model = deviceOps.getModel()
print('AndroidId: ' .. tostring(androidId))
print('MacAddress: ' .. tostring(macAddr))
print('製造商/型號: ' .. manufacturer .. ' / ' .. model)

-- D. 判斷是否平板 / 模擬器
print('是否平板: ' .. tostring(deviceOps.isTablet()))
print('是否模擬器: ' .. tostring(deviceOps.isEmulator()))

-- E. 取得唯一設備 ID（預設策略）
local uniqueId = deviceOps.getUniqueDeviceId()
print('UniqueDeviceId: ' .. tostring(uniqueId))

-- F. 取得唯一 ID（自訂前綴 + 是否強制重新生成）
local uniqueIdWithPrefix = deviceOps.getUniqueDeviceId('my_prefix')
local uniqueIdForce = deviceOps.getUniqueDeviceId(true)
local uniqueIdBoth = deviceOps.getUniqueDeviceId('my_prefix', true)
print('UniqueId(prefix): ' .. tostring(uniqueIdWithPrefix))
print('UniqueId(force): ' .. tostring(uniqueIdForce))
print('UniqueId(prefix+force): ' .. tostring(uniqueIdBoth))

-- G. 判斷是否與指定 ID 為同一台設備
local sameDevice = deviceOps.isSameDevice(uniqueId)
print('與 uniqueId 是否同一設備: ' .. tostring(sameDevice))

------------------------------------------------------------------------------------------

-- 取得 Phone 工具
local phoneOps = UtilCls.phoneOps()

-- A. 判斷是否為手機（部分平板可能 return false）
print('isPhone = ' .. tostring(phoneOps.isPhone()))

-- B. 取得設備與 SIM 資訊
print('DeviceId = ' .. tostring(phoneOps.getDeviceId()))
print('Serial = ' .. tostring(phoneOps.getSerial()))
-- print('IMEI = ' .. tostring(phoneOps.getIMEI() ))
-- print('MEID = ' .. tostring(phoneOps.getMEID()))
-- print('IMSI = ' .. tostring(phoneOps.getIMSI()))
print('PhoneType = ' .. tostring(phoneOps.getPhoneType()))
print('SIM Ready = ' .. tostring(phoneOps.isSimCardReady()))
print('SIM Operator Name = ' .. tostring(phoneOps.getSimOperatorName()))
print('SIM Operator MNC = ' .. tostring(phoneOps.getSimOperatorByMnc()))

-- C. 取得 ImeiOrMeid（true=IMEI，false=MEID）
local imeiOrMeid = phoneOps.getImeiOrMeid(true)
print('IMEI (manually choose) = ' .. tostring(imeiOrMeid))

-- D. 撥號 / 打電話
-- dial(): 只開啟撥號介面，不自動撥出
phoneOps.dial('10086')

-- call(): 直接撥出電話（需要 CALL_PHONE 權限）
-- phoneOps.call('10086')

-- E. 發送簡訊
-- 透過系統簡訊 App 發送，第二個參數為內容
phoneOps.sendSms('10086', '查詢餘額')

------------------------------------------------------------------------------------------

-- 2. 拿到 ScreenOps
local screenOps = UtilCls.screenOps()

-- 3. 讀取螢幕資訊
print('螢幕寬: ' .. tostring(screenOps.getScreenWidth()))
print('螢幕高: ' .. tostring(screenOps.getScreenHeight()))
print('DPI: ' .. tostring(screenOps.getScreenDensityDpi()))
print('是否直向: ' .. tostring(screenOps.isPortrait()))

-- 推荐用这种方式存储键值对

local spStaticOps = UtilCls.spStaticOps()

spStaticOps.putString('user_name', 'feiyun')

spStaticOps.putLong('last_login_time', os.time() * 1000)

local spOps = UtilCls.spOps()

lua
import('com.nx.assist.lua.LuaEngine')

local loader = LuaEngine.loadApk('xfxPlugin-release.apk'); assert(loader)
local UtilCls = loader.loadClass('com.xfx.plugin.UtilsMain'); assert(UtilCls, '插件入口类未找到')
UtilCls.init(LuaEngine.getContext())

------------------------------------------------------------------------------------------

local encryptOps = UtilCls.encryptOps()

-- 获取MD5
local md5 = encryptOps.encryptMD5ToString('hello world')
print('MD5 = ' .. tostring(md5))
-- 简略接口
print(encryptOps.md5('hello world'))

-- 帶鹽值版本
local md5Salt = encryptOps.encryptMD5ToString('hello world', 'my_salt')
print('MD5 with salt = ' .. tostring(md5Salt))

------------------------------------------------------------------------------------------


local timeOps = UtilCls.timeOps()

-- 格式化时间戳
local nowMillis = timeOps.getNowMills()
local fmt = timeOps.millis2String(nowMillis, 'yyyy-MM-dd HH:mm:ss')
print('現在時間: ' .. tostring(fmt))


-- 获取字符串工具
local strOps = UtilCls.stringOps()

-- 提取字串中的所有網址，返回 JSON 陣列字串
local text = '官方網站 https://www.example.com 以及部落格 http://blog.example.com/page?id=1'
local urlsJson = strOps.extractUrlsAsJson(text)
print(jsonLib.decode(urlsJson))


-- 分割字符串
local text = 'a,b,c'
local jsonArr = strOps.splitToJsonArray(text, ',')
print('json = ' .. tostring(jsonArr))

-- 解析成 Lua table（假設有 jsonLib）
-- local arr = jsonLib.decode(jsonArr)
-- for i, v in ipairs(arr) do
--     print(i, v)
-- end

-- 整数字符串转整数
local json = '{"a":"50","b":"80","c":"60"}'
