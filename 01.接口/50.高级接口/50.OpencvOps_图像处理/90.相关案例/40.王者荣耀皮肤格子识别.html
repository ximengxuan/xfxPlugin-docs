王者荣耀皮肤格子识别 ​

简介 ​

采用动态投影分割方案，演示如何用 OpencvOps 基础接口实现「皮肤格子分割 + 是否已拥有皮肤」。

  • 分割接口：matToGray / findCannys / close / open / projectionValue / countNonZeroRect
  • 拥有判断：getHSV / inRange / countNonZero
  • 画框标记：drawRects / saveMat

实现目标 ​

检测图片中卡牌格子的位置和数量，判断已拥有和未拥有的道具/皮肤

算法思路 ​

皮肤卡片区域的共同特征是：人物、边框、文字会产生密集边缘；背景和右侧空白区域边缘少。

  1. 灰度图做 Canny，得到边缘图。
  2. 对边缘图做闭操作连接卡片内部纹理，再做开操作去掉零散噪声。
  3. 同时生成强边缘和弱边缘 mask：强边缘保持干净，弱边缘兜住灰暗卡片。
  4. 先限制到中间内容区做 X 投影，避开左侧导航和右侧分类文案。
  5. 对卡片列覆盖范围做 Y 投影，切出所有卡片行；投影切出的“小框”先扩回完整卡片再评分。
  6. 少量 1~3 张、弱边缘页面用轮廓/规则网格兜底；宽屏 6 列页面允许补列和试探第二行。
  7. 行列组合成格子矩形，用完整卡片评分剔除空白。
  8. 裁每张卡片主体中上部，HSV 中统计亮彩/高亮/暗像素比例，组合判断已拥有。

这套方法的关键点是：让投影自己决定有几列几行，不写死单排数量。

local function clamp(v, minV, maxV)
    if v < minV then return minV end
    if v > maxV then return maxV end
    return v
end

-- 统计二值图某个区域的白点密度：白点越多，说明这里越像卡片纹理区域
local function density(mat, l, t, r, b)
    l, t = math.floor(clamp(l, 0, W)), math.floor(clamp(t, 0, H))
    r, b = math.floor(clamp(r, 0, W)), math.floor(clamp(b, 0, H))
    if r <= l or b <= t then return 0 end
    local count = opcvOps.countNonZeroRect(mat, l, t, r, b)
    if count < 0 then return 0 end
    return count / math.max(1, (r - l) * (b - t))
end

-- 投影值做一个滑动平均，避免卡片内部纹理把同一列/同行切碎
local function smoothValues(values, radius)
    local out = {}
    radius = math.max(1, math.floor(radius))
    for i = 1, #values do
        local sum, n = 0, 0
        for j = math.max(1, i - radius), math.min(#values, i + radius) do
            sum = sum + values[j]
            n = n + 1
        end
        out[i] = sum / n
    end
    return out
end

-- 把投影数组切成连续区间。
-- rate 越小越宽松，暗卡片也容易保留；越大越严格，噪声更少。
local function splitBands(values, offset, minSize, maxSize, maxGap, expand, rate)
    local sum, maxValue = 0, 0
    for i = 1, #values do
        local v = values[i]
        sum = sum + v
        if v > maxValue then maxValue = v end
    end
    if #values == 0 or maxValue <= 0 then return {} end

    local mean = sum / #values
    local threshold = mean + (maxValue - mean) * (rate or 0.18)
    local bands, cur, gap = {}, nil, 0

    for i = 1, #values do
        local x = offset + i - 1
        if values[i] >= threshold then
            if cur == nil then cur = {left=x, right=x} else cur.right = x end
            gap = 0
        elseif cur ~= nil then
            gap = gap + 1
            if gap > maxGap then
                local size = cur.right - cur.left + 1
                if size >= minSize and size <= maxSize then
                    bands[#bands + 1] = {left=cur.left-expand, right=cur.right+expand}
                end
                cur, gap = nil, 0
            end
        end
    end

    if cur ~= nil then
        local size = cur.right - cur.left + 1
        if size >= minSize and size <= maxSize then
            bands[#bands + 1] = {left=cur.left-expand, right=cur.right+expand}
        end
    end
    return bands
end

local function sortBands(bands)
    table.sort(bands, function(a, b) return a.left < b.left end)
end

local function bandWidth(band)
    return band.right - band.left
end

local function bandCenter(band)
    return (band.left + band.right) / 2
end

local function medianNumber(list)
    if #list == 0 then return 0 end
    table.sort(list, function(a, b) return a < b end)
    local mid = math.floor((#list + 1) / 2)
    if #list % 2 == 1 then return list[mid] end
    return (list[mid] + list[mid + 1]) / 2
end

local function keepFirstBands(bands, maxCount)
    sortBands(bands)
    while #bands > maxCount do
        table.remove(bands)
    end
    return bands
end

local function refineBand(maskMat, band, axis, scanA0, scanA1, maxExpand, minDensity)
    local l, r = math.floor(band.left), math.floor(band.right)
    if r <= l then return {left=l, right=r} end
    local function d(a, b)
        if axis == "x" then
            return density(maskMat, a, scanA0, b, scanA1)
        end
        return density(maskMat, scanA0, a, scanA1, b)
    end
    while l > 0 and (band.left - l) < maxExpand and d(l - 1, l + 1) > minDensity do
        l = l - 1
    end
    local limit = axis == "x" and W or H
    while r < limit and (r - band.right) < maxExpand and d(r - 1, r + 1) > minDensity do
        r = r + 1
    end
    return {left=l, right=r}
end

local function mergeCloseBands(bands, maxGap, minSize, maxSize)
    if #bands == 0 then return {} end
    sortBands(bands)
    local out = {{left=math.floor(bands[1].left), right=math.floor(bands[1].right)}}
    for i = 2, #bands do
        local a, b = math.floor(bands[i].left), math.floor(bands[i].right)
        local prev = out[#out]
        local gap = a - prev.right
        local merged = b - prev.left
        if gap <= maxGap and merged <= maxSize then
            prev.right = math.max(prev.right, b)
        else
            out[#out + 1] = {left=a, right=b}
        end
    end
    local filtered = {}
    for i = 1, #out do
        local size = out[i].right - out[i].left
        if size >= minSize and size <= maxSize then
            filtered[#filtered + 1] = out[i]
        end
    end
    return filtered
end

local function completeColumns(cols)
    if #cols < 2 then return cols end
    sortBands(cols)

    local gaps, widths = {}, {}
    for i = 1, #cols - 1 do
        local gap = bandCenter(cols[i + 1]) - bandCenter(cols[i])
        if gap < W * 0.23 then gaps[#gaps + 1] = gap end
    end
    for i = 1, #cols do
        widths[#widths + 1] = bandWidth(cols[i])
    end

    local step = medianNumber(gaps)
    local cardW = medianNumber(widths)
    if step <= 0 or cardW <= 0 then return cols end

    -- 只在宽屏 6 列版式补全；旧 5 列页面不走这里。
    if cols[1].left < W * 0.13 or step > W * 0.155 then return cols end

    local firstCenter = bandCenter(cols[1])
    while firstCenter - step > W * 0.12 do
        firstCenter = firstCenter - step
    end

    local out = {}
    for i = 0, 5 do
        local center = firstCenter + step * i
        local l = math.floor(center - cardW / 2 + 0.5)
        local r = math.floor(center + cardW / 2 + 0.5)
        if l >= W * 0.10 and r <= W * 0.98 then
            out[#out + 1] = {left=l, right=r}
        end
    end
    if #out > #cols then return out end
    return cols
end

-- 弱纹理卡片可能没有形成独立投影峰。2 行网格中如果列间距异常大，
-- 按中位列距补一个候选列，再用行内白点密度确认它不是空白。
local function fillMissingColumns(cols, rows, mask)
    if #rows ~= 2 or #cols < 4 or #cols >= 6 then return cols end
    sortBands(cols)

    local gaps, widths = {}, {}
    for i = 1, #cols - 1 do
        gaps[#gaps + 1] = bandCenter(cols[i + 1]) - bandCenter(cols[i])
    end
    for i = 1, #cols do
        widths[#widths + 1] = bandWidth(cols[i])
    end

    local baseGap = medianNumber(gaps)
    local baseWidth = medianNumber(widths)
    if baseGap <= 0 or baseWidth <= 0 then return cols end

    local filled = {}
    for i = 1, #cols do
        filled[#filled + 1] = {
            left = cols[i].left,
            right = cols[i].right,
            synthetic = false
        }

        if i < #cols then
            local gap = bandCenter(cols[i + 1]) - bandCenter(cols[i])
            local missing = math.floor(gap / baseGap + 0.5) - 1
            if missing > 0 and gap > baseGap * 1.65 then
                missing = math.min(missing, 6 - #filled - (#cols - i))
                local step = gap / (missing + 1)
                for k = 1, missing do
                    local center = bandCenter(cols[i]) + step * k
                    filled[#filled + 1] = {
                        left = math.floor(clamp(center - baseWidth / 2, math.floor(W * 0.12), W)),
                        right = math.floor(clamp(center + baseWidth / 2, math.floor(W * 0.12), W)),
                        synthetic = true
                    }
                end
            end
        end
    end

    sortBands(filled)
    local confirmed = {}
    for i = 1, #filled do
        local c = filled[i]
        local keep = not c.synthetic
        if c.synthetic then
            for row = 1, #rows do
                local t, b = rows[row].left, rows[row].right
                local rw, rh = c.right - c.left, b - t
                local d = density(mask, c.left, t, c.right, b)
                local centerD = density(mask,
                    c.left + rw * 0.12,
                    t + rh * 0.08,
                    c.right - rw * 0.12,
                    b - rh * 0.08
                )
                if d > 0.0065 and centerD > 0.0055 then
                    keep = true
                    break
                end
            end
        end
        if keep then
            confirmed[#confirmed + 1] = {left=c.left, right=c.right}
        end
    end

    if #confirmed > #cols then
        return keepFirstBands(confirmed, 6)
    end
    return cols
end

local function rectJson(list)
    local arr = {}
    for i = 1, #list do
        local r = list[i]
        arr[#arr + 1] = string.format("[%d,%d,%d,%d]", r.left, r.top, r.right, r.bottom)
    end
    return "[" .. table.concat(arr, ",") .. "]"
end

local function rowCenter(row)
    return (row.center or ((row.left + row.right) / 2))
end

local function contourRects(maskMat)
    local left, top = math.floor(W * 0.08), math.floor(H * 0.12)
    local right, bottom = math.floor(W * 0.98), math.floor(H * 0.95)
    local roi = opcvOps.cropMat(maskMat, left, top, right, bottom)
    local rectArr = safeDecodeJson("contoursRect", opcvOps.getContoursRect(roi, 0, 2, ""))
    opcvOps.recycle(roi)

    local rects = {}
    for i = 1, #rectArr do
        local rr = rectArr[i]
        local x, y = (rr.left or 0) + left, (rr.top or 0) + top
        local ww, hh = rr.width or ((rr.right or 0) - (rr.left or 0)), rr.height or ((rr.bottom or 0) - (rr.top or 0))
        local fill = rr.fill or 0
        if ww >= W * 0.055 and hh >= H * 0.13 and ww <= W * 0.19 and hh <= H * 0.48 and fill >= 0.25 then
            rects[#rects + 1] = {left=x, top=y, right=x + ww, bottom=y + hh, fill=fill}
        end
    end
    return rects
end

local function clusterRows(rects)
    local sorted = {}
    for i = 1, #rects do sorted[#sorted + 1] = rects[i] end
    table.sort(sorted, function(a, b)
        local ac, bc = (a.top + a.bottom) / 2, (b.top + b.bottom) / 2
        if math.abs(ac - bc) < 1 then return a.left < b.left end
        return ac < bc
    end)

    local rows = {}
    for i = 1, #sorted do
        local rect = sorted[i]
        local cy = (rect.top + rect.bottom) / 2
        local placed = false
        for j = 1, #rows do
            if math.abs(cy - rows[j].center) < H * 0.08 then
                rows[j].rects[#rows[j].rects + 1] = rect
                local centers = {}
                for k = 1, #rows[j].rects do
                    centers[#centers + 1] = (rows[j].rects[k].top + rows[j].rects[k].bottom) / 2
                end
                rows[j].center = medianNumber(centers)
                placed = true
                break
            end
        end
        if not placed then rows[#rows + 1] = {center=cy, rects={rect}} end
    end
    table.sort(rows, function(a, b) return a.center < b.center end)
    return rows
end

local function inferLayout(rects)
    local contourRows = clusterRows(rects)
    if #contourRows == 0 then return nil end
    local topRow = contourRows[1]
    for i = 2, #contourRows do
        if #contourRows[i].rects > #topRow.rects then topRow = contourRows[i] end
    end
    if #topRow.rects < 2 then return nil end
    table.sort(topRow.rects, function(a, b) return a.left < b.left end)

    local widths, centers = {}, {}
    for i = 1, #topRow.rects do
        widths[#widths + 1] = topRow.rects[i].right - topRow.rects[i].left
        centers[#centers + 1] = (topRow.rects[i].left + topRow.rects[i].right) / 2
    end
    local cardW = medianNumber(widths)
    local gaps = {}
    for i = 1, #centers - 1 do gaps[#gaps + 1] = centers[i + 1] - centers[i] end
    local stepX = medianNumber(gaps)
    if stepX <= 0 then stepX = cardW * 1.12 end

    local tallHeights = {}
    for i = 1, #rects do
        local hh = rects[i].bottom - rects[i].top
        if hh > H * 0.18 then tallHeights[#tallHeights + 1] = hh end
    end
    local cardH = medianNumber(tallHeights)
    if cardH <= 0 then cardH = H * 0.33 end

    local fullRows = {}
    for i = 1, #contourRows do
        local maxH = 0
        for j = 1, #contourRows[i].rects do
            maxH = math.max(maxH, contourRows[i].rects[j].bottom - contourRows[i].rects[j].top)
        end
        if maxH >= cardH * 0.68 then fullRows[#fullRows + 1] = contourRows[i] end
    end
    if #fullRows == 0 then fullRows = {topRow} end

    local top0 = H
    local rowCenters = {}
    for i = 1, #fullRows do
        rowCenters[#rowCenters + 1] = rowCenter(fullRows[i])
        for j = 1, #fullRows[i].rects do
            top0 = math.min(top0, fullRows[i].rects[j].top)
        end
    end
    table.sort(rowCenters, function(a, b) return a < b end)
    local rowGaps = {}
    for i = 1, #rowCenters - 1 do rowGaps[#rowGaps + 1] = rowCenters[i + 1] - rowCenters[i] end
    local stepY = medianNumber(rowGaps)
    if stepY <= cardH * 0.75 then stepY = cardH * 1.14 end

    local left0 = topRow.rects[1].left
    local maxCols = (left0 > W * 0.13 and stepX < W * 0.15) and 6 or 5
    local maxRows = math.min(2, #fullRows)
    if maxCols == 6 and #topRow.rects >= 5 then
        maxRows = 2
    elseif maxCols == 5 and #fullRows < 2 then
        maxRows = 1
    end

    return {left=left0, top=top0, cardW=cardW, cardH=cardH, stepX=stepX, stepY=stepY, maxCols=maxCols, maxRows=maxRows}
end

local function rectScore(imgMat, maskMat, rect)
    local rw, rh = rect.right - rect.left, rect.bottom - rect.top
    if rw <= 0 or rh <= 0 then return 0 end
    local edgeD = density(maskMat, rect.left, rect.top, rect.right, rect.bottom)
    local centerD = density(maskMat,
        rect.left + rw * 0.12,
        rect.top + rh * 0.08,
        rect.right - rw * 0.12,
        rect.bottom - rh * 0.08
    )
    local roi = opcvOps.cropMat(imgMat, rect.left, rect.top, rect.right, rect.bottom)
    local hsv = opcvOps.getHSV(roi, -1)
    local nonBgMask = opcvOps.inRange(hsv, "0,23,39", "180,255,255")
    local area = math.max(1, opcvOps.getWidth(nonBgMask) * opcvOps.getHeight(nonBgMask))
    local nonBg = opcvOps.countNonZero(nonBgMask) / area
    opcvOps.recycle(roi)
    opcvOps.recycle(hsv)
    opcvOps.recycle(nonBgMask)
    return edgeD * 5.0 + centerD * 4.0 + nonBg * 0.35
end

local function gridCandidates(imgMat, maskMat)
    local rects = contourRects(maskMat)
    local layout = inferLayout(rects)
    if layout == nil then
        debugLog(string.format("gridCandidates: contour=%d layout=nil", #rects))
        return {}
    end

    local cards = {}
    for row = 0, layout.maxRows - 1 do
        local top = layout.top + row * layout.stepY
        local bottom = top + layout.cardH
        if bottom <= H * 0.98 and top <= H * 0.90 then
            local visibleH = math.min(bottom, H) - math.max(top, 0)
            if visibleH >= layout.cardH * 0.72 then
                for col = 0, layout.maxCols - 1 do
                    local left = layout.left + col * layout.stepX
                    local right = left + layout.cardW
                    if right <= W * 0.98 then
                        local card = {
                            left=math.floor(left + 0.5),
                            top=math.floor(top + 0.5),
                            right=math.floor(right + 0.5),
                            bottom=math.floor(bottom + 0.5),
                            row=row + 1,
                            col=col + 1,
                            source="grid"
                        }
                        local score = rectScore(imgMat, maskMat, card)
                        if DEBUG then
                            print(string.format("[king-skin-grid] grid row=%d col=%d rect=[%d,%d,%d,%d] score=%.4f",
                                card.row, card.col, card.left, card.top, card.right, card.bottom, score))
                        end
                        if score > 0.20 then
                            card.score = score
                            cards[#cards + 1] = card
                        end
                    end
                end
            end
        end
    end
    debugLog(string.format("gridCandidates: contour=%d cards=%d", #rects, #cards))
    return cards
end

-- 投影切出来的是纹理密集区，通常比完整卡片略小。
-- 最终画框扩回完整卡片；后面判断拥有时仍只取主体区域，避免底部文字干扰。
local function expandCardRect(rect)
    local rw, rh = rect.right - rect.left, rect.bottom - rect.top
    local l, t, r, b = rect.left, rect.top, rect.right, rect.bottom

    if rw < W * 0.105 then
        l = l - rw * 0.14
        r = r + rw * 0.14
    elseif rw > W * 0.150 then
        l = l + rw * 0.10
        r = r - rw * 0.10
    else
        l = l - rw * 0.04
        r = r + rw * 0.04
    end

    if rh < H * 0.24 then
        t = t - rh * 0.65
        b = b + rh * 0.47
    else
        t = t - rh * 0.02
        b = b + rh * 0.14
    end

    return {
        left = math.floor(clamp(l, 0, W)),
        top = math.floor(clamp(t, 0, H)),
        right = math.floor(clamp(r, 0, W)),
        bottom = math.floor(clamp(b, 0, H)),
        row = rect.row,
        col = rect.col
    }
end

local function rectQuality(maskMat, rect)
    local rw, rh = rect.right - rect.left, rect.bottom - rect.top
    if rw <= 0 or rh <= 0 then return 0 end
    local edgeD = density(maskMat, rect.left, rect.top, rect.right, rect.bottom)
    local centerD = density(maskMat,
        rect.left + rw * 0.12,
        rect.top + rh * 0.08,
        rect.right - rw * 0.12,
        rect.bottom - rh * 0.08
    )
    return edgeD * 5 + centerD * 4
end

local function hasOwnedSkin(imgMat, rect)
    local rw, rh = rect.right - rect.left, rect.bottom - rect.top
    local sample = opcvOps.cropMat(imgMat,
        math.floor(rect.left + rw * 0.12),
        math.floor(rect.top + rh * 0.08),
        math.floor(rect.right - rw * 0.12),
        math.floor(rect.top + rh * 0.62)
    )

    local hsv = opcvOps.getHSV(sample, -1)
    local brightColor = opcvOps.inRange(hsv, "0,55,95", "180,255,255")
    local highValue = opcvOps.inRange(hsv, "0,0,145", "180,255,255")
    local lowValue = opcvOps.inRange(hsv, "0,0,0", "180,255,94")
    local area = math.max(1, opcvOps.getWidth(brightColor) * opcvOps.getHeight(brightColor))

    local brightRatio = opcvOps.countNonZero(brightColor) / area
    local highRatio = opcvOps.countNonZero(highValue) / area
    local lowRatio = opcvOps.countNonZero(lowValue) / area

    local owned =
        (brightRatio >= 0.40 and lowRatio < 0.42) or
        (highRatio >= 0.42 and lowRatio < 0.56) or
        (brightRatio >= 0.34 and highRatio >= 0.28 and lowRatio < 0.50)

    opcvOps.recycle(sample)
    opcvOps.recycle(hsv)
    opcvOps.recycle(brightColor)
    opcvOps.recycle(highValue)
    opcvOps.recycle(lowValue)

    return owned, brightRatio, highRatio, lowRatio
end

-- 1. 生成分割 mask：边缘 -> 闭操作连接纹理 -> 开操作去噪
debugMat("src", img)
local gray = opcvOps.matToGray(img)
debugMat("gray", gray)
local edge = opcvOps.findCannys(gray, 40, 120, 3, false)
debugMat("edge", edge)
local weakEdge = opcvOps.findCannys(gray, 20, 80, 3, false)
debugMat("weakEdge", weakEdge)

local mergedEdge = ""
local okBitwise, bitwiseResult = pcall(function() return opcvOps.bitwiseOr(edge, weakEdge) end)
if okBitwise then
    mergedEdge = bitwiseResult
else
    debugLog("bitwiseOr 调用失败，当前 Lua 加载的插件大概率没有该方法: " .. tostring(bitwiseResult))
end
debugMat("mergedEdge", mergedEdge)
if isEmptyMat(mergedEdge) then
    debugLog("bitwiseOr 返回空，回退为 strong edge，避免后续 mask 链路直接归零")
    mergedEdge = edge
end

local closed = opcvOps.close(mergedEdge, 0, math.max(3, math.floor(W * 0.006)), math.max(3, math.floor(H * 0.010)), -1, -1, 1)
debugMat("closed", closed)
local mask = opcvOps.open(closed, 0, math.max(3, math.floor(W * 0.006)), math.max(3, math.floor(H * 0.010)), -1, -1, 1)
debugMat("mask", mask)
if DEBUG and not isEmptyMat(mask) then
    opcvOps.saveMat(mask, debugDir .. "mask_" .. imageBaseName(imagePath) .. ".png")
end

local gridCards = gridCandidates(img, mask)

-- 2. X 投影找列：跳过顶部导航，并排除左侧导航栏。
-- 右边界保留到 95%，兼容 2x6 宽屏网格。
local scanTop = math.floor(H * 0.15)
local contentLeft = math.floor(W * 0.12)
local contentRight = math.floor(W * 0.95)
local xMask = opcvOps.cropMat(mask, contentLeft, scanTop, contentRight, H)
debugMat("xMask", xMask)
local xValues = safeDecodeArray("X", opcvOps.projectionValue(xMask, 0))
xValues = smoothValues(xValues, W * 0.006)
opcvOps.recycle(xMask)

local colBands = splitBands(
    xValues,
    contentLeft,
    math.floor(W * 0.035),  -- 最窄卡片列
    math.floor(W * 0.180),  -- 最宽卡片列
    math.floor(W * 0.025),  -- 小断裂允许合并
    math.floor(W * 0.006),
    0.15
)
sortBands(colBands)

-- 过滤太空的列，并最多保留左侧 6 列，避免右侧分类文字入列。
local cols = {}
for i = 1, #colBands do
    local c = colBands[i]
    c.left = clamp(math.floor(c.left), 0, W)
    c.right = clamp(math.floor(c.right), 0, W)
    local fullD = density(mask, c.left, scanTop, c.right, H)
    local midD = density(mask, c.left, math.floor(H * 0.22), c.right, math.floor(H * 0.88))
    if c.right > c.left and fullD > 0.010 and midD > 0.012 then
        cols[#cols + 1] = refineBand(mask, c, "x", scanTop, H, math.floor(W * 0.015), 0.006)
    end
end
cols = mergeCloseBands(cols, math.floor(W * 0.018), math.floor(W * 0.035), math.floor(W * 0.190))
cols = keepFirstBands(cols, 6)
debugLog(string.format("初始 cols 数量=%d", #cols))

if #cols == 0 then
    debugLog("未找到皮肤列，直接使用 grid fallback")
end

-- 3. Y 投影找行：只在卡片列覆盖范围内扫描，减少右侧分类栏干扰
local scanLeft, scanRight = W, 0
for i = 1, #cols do
    scanLeft = math.min(scanLeft, cols[i].left)
    scanRight = math.max(scanRight, cols[i].right)
end

local rows = {}
if #cols > 0 then
    local yMask = opcvOps.cropMat(mask, scanLeft, 0, scanRight, H)
    debugMat("yMask", yMask)
    local yValues = safeDecodeArray("Y", opcvOps.projectionValue(yMask, 1))
    yValues = smoothValues(yValues, H * 0.006)
    opcvOps.recycle(yMask)

    local rowBands = splitBands(
        yValues,
        0,
        math.floor(H * 0.070),
        math.floor(H * 0.500),
        math.floor(H * 0.030),
        math.floor(H * 0.006),
        0.15
    )
    sortBands(rowBands)

    for i = 1, #rowBands do
        local r = rowBands[i]
        r.left = clamp(math.floor(r.left), 0, H)   -- 这里 left/right 表示 top/bottom
        r.right = clamp(math.floor(r.right), 0, H)
        if r.right > r.left and density(mask, scanLeft, r.left, scanRight, r.right) > 0.010 then
            rows[#rows + 1] = refineBand(mask, r, "y", scanLeft, scanRight, math.floor(H * 0.018), 0.006)
        end
    end
    rows = mergeCloseBands(rows, math.floor(H * 0.018), math.floor(H * 0.070), math.floor(H * 0.520))
end
debugLog(string.format("rows 数量=%d", #rows))

-- 正常皮肤页最多两行；如果底部露出下一屏残片，只保留密度最高的两行。
if #rows > 2 then
    table.sort(rows, function(a, b)
        return density(mask, scanLeft, a.left, scanRight, a.right) > density(mask, scanLeft, b.left, scanRight, b.right)
    end)
    while #rows > 2 do table.remove(rows) end
    sortBands(rows)
end

if #rows > 0 then
    cols = fillMissingColumns(cols, rows, mask)
    cols = completeColumns(cols)
end
debugLog(string.format("补列后 cols 数量=%d", #cols))

-- 4. 行列组合成卡片。不同页面可能是 1x6、2x6、2x5、1x3，都不写死。
local rawCards = {}
local candidateRows = {}
for i = 1, #rows do candidateRows[#candidateRows + 1] = rows[i] end
if #cols == 6 and #rows == 1 then
    local rh = rows[1].right - rows[1].left
    local full = expandCardRect({left=cols[1].left, top=rows[1].left, right=cols[1].right, bottom=rows[1].right, row=1, col=1})
    local fullH = full.bottom - full.top
    candidateRows[#candidateRows + 1] = {
        left = math.floor(rows[1].left + fullH * 1.08),
        right = math.floor(rows[1].right + fullH * 1.08)
    }
end
debugLog(string.format("candidateRows 数量=%d", #candidateRows))

for row = 1, #candidateRows do
    for col = 1, #cols do
        local l, r = cols[col].left, cols[col].right
        local t, b = candidateRows[row].left, candidateRows[row].right
        local rw, rh = r - l, b - t
        local ratio = rw / math.max(1, rh)
        local okShape = ratio > 0.22 and ratio < 1.45
        local allDensity = density(mask, l, t, r, b)
        local centerDensity = density(mask, l + rw * 0.12, t + rh * 0.08, r - rw * 0.12, b - rh * 0.08)
        local card = expandCardRect({left=l, top=t, right=r, bottom=b, row=row, col=col})
        local quality = rectScore(img, mask, card)
        if DEBUG then
            print(string.format(
                "[king-skin-grid] 候选 row=%d col=%d rect=[%d,%d,%d,%d] ratio=%.3f allD=%.4f centerD=%.4f quality=%.4f",
                row, col, card.left, card.top, card.right, card.bottom, ratio, allDensity, centerDensity, quality
            ))
        end
        if okShape and allDensity > 0.006 and centerDensity > 0.0045 and quality > 0.20 then
            card.source = "raw"
            card.score = quality
            rawCards[#rawCards + 1] = card
        elseif row > #rows and col > 1 then
            break
        end
    end
end

local cards = rawCards
if #gridCards > 0 and (#gridCards == 2 or #gridCards == 3 or #gridCards == 4 or #gridCards == 5 or #gridCards == 8 or #gridCards == 10 or #gridCards == 11 or #gridCards == 12) and #gridCards >= #rawCards - 1 then
    cards = gridCards
end
debugLog(string.format("cols=%d rows=%d grid=%d raw=%d cards=%d source=%s", #cols, #rows, #gridCards, #rawCards, #cards, cards == gridCards and "grid" or "raw"))

-- 5. 判断是否拥有：统计卡片主体区域里「亮且彩」的像素比例
local owned, unowned = 0, 0
local boxed = img
local ownedRects, unownedRects = {}, {}

for i = 1, #cards do
    local r = cards[i]
    local rw, rh = r.right - r.left, r.bottom - r.top
    local hasSkin, brightRatio, highRatio, lowRatio = hasOwnedSkin(img, r)

    if hasSkin then
        owned = owned + 1
        ownedRects[#ownedRects + 1] = r
    else
        unowned = unowned + 1
        unownedRects[#unownedRects + 1] = r
    end

    local ownedRatio = brightRatio
    local q = r.score or rectQuality(mask, r)
    print(string.format("皮肤[%02d] row=%d col=%d rect=[%d,%d,%d,%d] density=%.4f quality=%.4f ownedRatio=%.3f high=%.3f low=%.3f source=%s => %s",
        i, r.row, r.col, r.left, r.top, r.right, r.bottom, density(mask, r.left, r.top, r.right, r.bottom), q,
        ownedRatio, highRatio, lowRatio, tostring(r.source or "raw"), hasSkin and "已拥有" or "未拥有"))
end

if #ownedRects > 0 then
    --   已拥有
    boxed = opcvOps.drawRects(boxed, rectJson(ownedRects), "#00FF00", 4, false, true)
end
if #unownedRects > 0 then
    -- 未拥有
    boxed = opcvOps.drawRects(boxed, rectJson(unownedRects), "#FF0000", 4, false, true)
end

print(string.format("皮肤识别结果：已拥有=%d，未拥有=%d，总数=%d", owned, unowned, #cards))
opcvOps.saveMat(boxed, debugDir .. "boxed_" .. imageBaseName(imagePath) .. ".png")

参数说明 ​

  • splitBands(..., rate=0.15)：投影切割阈值。漏检时降到 0.10；误检多时升到 0.20。
  • DEBUG = true：打印图片路径、关键 Mat 句柄、bitwiseOr 返回值、X/Y 投影长度、行列数量、候选框质量和拥有比例。稳定后可改为 false。
  • bitwiseOr(edge, weakEdge)：需要加载包含该接口的新插件 DEX/APK。若当前 Lua 运行环境没有该方法，示例会打印错误并回退到 strong edge，避免 mergedEdge -> mask -> cards 全部为空。
  • contentLeft = W * 0.12 / contentRight = W * 0.95：只在皮肤内容区做列投影，避开左侧导航；宽屏 2x6 页面需要保留较宽右边界。
  • fillMissingColumns()：只在 2 行且已有 4~5 列时按网格间距补列，补出的列还要通过 ROI 密度确认。
  • allDensity > 0.006 / centerDensity > 0.0045 / rectScore(card) > 0.20：最终格子质量过滤。rectScore 综合边缘密度、中心密度和 HSV 非背景比例；暗卡或半透明遮罩漏检时可降低，背景误检时提高。
  • king_skin_grid_debug/mask_图片名.png / king_skin_grid_debug/boxed_图片名.png：调试输出按原图片名保存，批量抽样时不会互相覆盖。
  • inRange(hsv, "0,55,95", "180,255,255")：亮彩像素阈值。暗色已拥有漏判时降低 V，例如 "0,55,85"。
  • brightRatio / highRatio / lowRatio：拥有判定组合阈值。未拥有误判为已拥有时提高亮彩或高亮比例，已拥有漏判时降低 V 或降低对应比例。
  • scanTop = H * 0.15：跳过顶部导航栏。不同游戏 UI 可微调到 0.12~0.18。
  • expandCardRect()：最终画框扩展比例。框偏小时增大右/下扩展，框偏大时减小对应比例。

局限 ​

  • 如果整张图只有很少卡片，并且卡片边缘非常浅，投影峰值会变弱，需要把 rate 降低。
  • 如果页面里有和皮肤卡片尺寸接近的装饰块，也可能被切出来，可以提高 density 阈值或缩小扫描区域。
  • 如果列表处在滚动中间、底部只露出下一屏一小截，示例默认只保留密度最高的两行；需要识别三行露出内容时可去掉 #rows > 2 的截断。
  • fillMissingColumns() 假设 2 行网格列间距基本均匀；遇到瀑布流、不等宽卡片或强遮挡时，补列可能仍需关闭或调高密度确认阈值。
