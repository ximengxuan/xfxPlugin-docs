# 腾讯防水墙滑块识别

## 拼图滑块验证码定位（腾讯防水墙）

演示如何用 `OpencvOps` 基础接口实现"拼图滑块验证码"目标缺口定位。

### 适用场景

图片中横向排列 3 个拼图形状（360x361 分辨率最常见）：

* **最左**：空心/浅色描边的「源拼图」（要拖动的滑块图案）
* **中间 / 最右**：深色填充拼图（其中之一是目标缺口）

> 用基础方法组合定位，自动判断源拼图与哪个深色拼图的缺口形状匹配，取匹配者作为目标，返回**滑块需要拖动的水平像素距离**。

### 用到的接口

只用 `OpencvOps` 的基础方法：

| 接口 | 用途 |
|------|------|
| `bitmapToMat` / `readResMat` | Bitmap 或文件 → Mat 句柄 |
| `matToGray` | 灰度化 |
| `matThreshold` | 二值化（多阈值收集候选） |
| `open` | 开操作去噪 |
| `getContoursRect` | 轮廓外接矩形 + `fill` 填充率 |
| `cropMat` | 裁剪模板 |
| `findImg` | 模板匹配找图 |
| `drawRect` / `saveMat` | 画框 / 保存（可选） |

### 算法思路（5 步）

1. **灰度化 + 多次二值化收集候选**：OTSU 正反 + 固定阈值 80/100/120/140/160（共 12 种），找连通域，按 (x/5, y/5) 去重保留面积最大
2. **过滤候选**：尺寸 20~100、纵横比 0.4~2.5、Y ∈ \[60, 220]、填充率 ≥ 0.1
3. **找同 Y 行**：cy 中位数对齐（差 ≤ 15），按 cx 找 15% / 50% / 85% 三个等间距候选
4. **源拼图推断**：若最左候选 fill > 0.5 且 cx > 20% 宽，源拼图在 gap \* 0.55 处推断位置（轮廓过淡无法直接检测时也能定位）
5. **模板匹配确定目标**：取源拼图区域为模板，`findImg` 在中间/右侧候选区域分别搜索，取较高分者为目标

### Lua 示例

```lua
local UtilsMain = loader.loadClass('com.xfx.plugin.UtilsMain')
UtilsMain.init(LuaEngine.getContext())


-- 引入图像识别类
local opcvOps = XFX:getOps('opencvOps'); assert(opcvOps)

local saveDir = '/sdcard/Download/opencvTest/'

-- 1. 读取图片（或 LuaEngine.snapShot 截图）
-- local bmp = LuaEngine.snapShot(0, 0, 0, 0)
local srcMat = opcvOps.readResMat("/sdcard/Download/插件实例/20260805_110457_005.png")
if srcMat == "" then print("读取图片失败"); return end

local W = opcvOps.getWidth(srcMat)
local H = opcvOps.getHeight(srcMat)

-- 2. 灰度化
local grayMat = opcvOps.matToGray(srcMat)

-- 3. 多次二值化收集候选连通域
-- 类型：0=BINARY, 1=BINARY_INV, 8=OTSU, 9=OTSU_INV
-- OTSU 阈值传 0（OpenCV 自动计算）
local function tryThreshold(gray, thresh, maxval, type)
    return opcvOps.matThreshold(gray, thresh, maxval, type)
end

local binaryMats = {}
-- OTSU 正反
table.insert(binaryMats, tryThreshold(grayMat, 0, 255, 9))  -- OTSU_INV（深色前景）
table.insert(binaryMats, tryThreshold(grayMat, 0, 255, 8))  -- OTSU（浅色前景）
-- 固定阈值
for _, t in ipairs({80, 100, 120, 140, 160}) do
    table.insert(binaryMats, tryThreshold(grayMat, t, 255, 1))  -- BINARY_INV
    table.insert(binaryMats, tryThreshold(grayMat, t, 255, 0))  -- BINARY
end

-- 4. 收集候选 + 去重 + 过滤
local candidates = {}  -- 数组（每个元素是一个 rect 表）
local dedup = {}       -- 5px 位置去重 key -> rect 表
for _, binMat in ipairs(binaryMats) do
    if binMat ~= '' then
        -- 开操作去噪
        local opened = opcvOps.open(binMat, 0, 3, 3, -1, -1, 1)
        if opened ~= '' then
            local rectsJson = opcvOps.getContoursRect(opened, 0, 2, '')
            if rectsJson ~= '' then
                local rects = jsonLib.decode(rectsJson)
                for _, r in ipairs(rects) do
                    local w = r.right - r.left
                    local h = r.bottom - r.top
                    local cy = r.top + h // 2
                    -- Y 范围 [60, 220]：排除顶部 title 和底部按钮
                    if cy >= 60 and cy <= 220
                       and w >= 20 and w <= 100
                       and h >= 20 and h <= 100
                       and (w * 1.0 / h) >= 0.4 and (w * 1.0 / h) <= 2.5
                       and r.fill >= 0.1 then
                        local key = (r.left // 5) .. ',' .. (r.top // 5)
                        local existing = dedup[key]
                        if existing == nil or (existing.right - existing.left) * (existing.bottom - existing.top) < w * h then
                            r.cx = r.left + w // 2
                            r.cy = cy
                            dedup[key] = r
                        end
                    end
                end
            end
            opcvOps.recycle(opened)
        end
        opcvOps.recycle(binMat)
    end
end

-- 转数组
for _, v in pairs(dedup) do
    table.insert(candidates, v)
end

if #candidates < 3 then
    print('识别失败：候选连通域不足 (' .. #candidates .. ')')
    opcvOps.recycle(grayMat)
    opcvOps.recycle(srcMat)
    return
end

-- 5. 按 cx 排序，找同 Y 行
table.sort(candidates, function(a, b) return a.cx < b.cx end)

-- 计算 cy 中位数
local cyList = {}
for _, c in ipairs(candidates) do
    table.insert(cyList, c.cy)
end
table.sort(cyList)
local medianCy = cyList[#cyList // 2 + 1]

-- 同 Y 行候选（cy 差 ≤ 15）
local sameRow = {}
for _, c in ipairs(candidates) do
    if math.abs(c.cy - medianCy) <= 15 then
        table.insert(sameRow, c)
    end
end
if #sameRow < 3 then
    sameRow = candidates  -- 放宽
end

-- 6. 选 3 个等间距候选（15% / 50% / 85% 位置）
local function pickClosest(pool, targetCx)
    local best, bestDiff = nil, math.huge
    for _, c in ipairs(pool) do
        local diff = math.abs(c.cx - targetCx)
        if diff < bestDiff then
            bestDiff = diff
            best = c
        end
    end
    return best
end

local targets = { W * 0.15, W * 0.5, W * 0.85 }
local selected = {}
for _, tc in ipairs(targets) do
    local b = pickClosest(sameRow, tc)
    if b then
        local found = false
        for _, s in ipairs(selected) do
            if s == b then found = true; break end
        end
        if not found then table.insert(selected, b) end
    end
end
if #selected < 3 then
    -- 退化：取 cx 最小/中/大的 3 个
    table.sort(sameRow, function(a, b) return a.cx < b.cx end)
    selected = { sameRow[1], sameRow[math.ceil(#sameRow/2)], sameRow[#sameRow] }
end
table.sort(selected, function(a, b) return a.cx < b.cx end)

local leftmost, middle, right = selected[1], selected[2], selected[3]

-- 7. 源拼图推断：若最左 fill > 0.5，向左推断源拼图位置
local source = leftmost
local inferred = false
if leftmost.fill > 0.5 and leftmost.cx > W * 0.20 then
    local gap = right.cx - leftmost.cx
    local inferredCx = leftmost.cx - gap * 0.55
    local infW = leftmost.right - leftmost.left
    local infLeft = inferredCx - infW // 2
    if infLeft >= 5 and inferredCx > 5 then
        source = {
            left = infLeft, right = infLeft + infW,
            top = leftmost.top, bottom = leftmost.bottom,
            fill = 0.0, 
			cx = math.floor(inferredCx), 
			cy = math.floor(leftmost.cy)
        }
        inferred = true
        middle = leftmost  -- 重新分配：推断后的最左候选变成中间 dark
    end
end

-- 8. 提取源拼图区域为模板
local pad = 2
local tx0 = math.max(0, source.left - pad)
local ty0 = math.max(0, source.top - pad)
local tx1 = math.min(W, source.right + pad)
local ty1 = math.min(H, source.bottom + pad)
if tx1 - tx0 < 10 or ty1 - ty0 < 10 then
    -- fallback：源拼图太小，用 middle 的尺寸
    tx0 = math.max(0, middle.left - pad); ty0 = math.max(0, middle.top - pad)
    tx1 = math.min(W, middle.right + pad); ty1 = math.min(H, middle.bottom + pad)
end
local templateMat = opcvOps.cropMat(grayMat, tx0, ty0, tx1, ty1)

-- 9. 模板匹配：在 middle 和 right 区域分别搜索，取较高分
local function matchInRegion(cand)
    -- findImg(matId, templateMat, maxCount, precision, x1, y1, x2, y2)
    local sX1 = math.max(0, cand.left - 10)
    local sY1 = math.max(0, cand.top - 10)
    local sX2 = math.min(W, cand.right + 10)
    local sY2 = math.min(H, cand.bottom + 10)
    local res = opcvOps.findImg(grayMat, templateMat, 1, 0.0, sX1, sY1, sX2, sY2)
    if res == '[]' or res == '' then
        return -1, -1, -1
    end
    local matches = jsonLib.decode(res)
    if not matches or #matches == 0 then
        return -1, -1, -1
    end
    local best = matches[1]
    return best.score, best.centerX, best.centerY
end

local sMid, midCx, midCy = matchInRegion(middle)
local sRight, rightCx, rightCy = matchInRegion(right)

opcvOps.recycle(templateMat)

-- 10. 取较高分者为目标
local targetIdx, targetCx, targetCy, bestScore
if sMid >= sRight then
    targetIdx, targetCx, targetCy, bestScore = 0, midCx, midCy, sMid
else
    targetIdx, targetCx, targetCy, bestScore = 1, rightCx, rightCy, sRight
end

local sourceCx = source.cx
local sourceCy = source.cy
local distance = targetCx - sourceCx

print(sourceCx, sourceCy,distance)
print(string.format('源拼图: (%d, %d) [推断=%s]', sourceCx, sourceCy, tostring(inferred)))
print(string.format('目标缺口: (%d, %d) [索引=%d, 匹配分=%.3f]', targetCx, targetCy, targetIdx, bestScore))
print('滑动距离: ' .. distance .. ' 像素')

-- 11. 画框标注（可选）
local vis = opcvOps.drawRect(srcMat, source.left, source.top, source.right, source.bottom, '#FF0000', 2, false, true)   -- 源位置
vis = opcvOps.drawRect(vis, middle.left, middle.top, middle.right, middle.bottom, '#00FFFF', 1, false, true)
vis = opcvOps.drawRect(vis, right.left, right.top, right.right, right.bottom, '#FF00FF', 1, false, true)         
vis = opcvOps.drawRect(vis, targetCx - 5, targetCy - 5, targetCx + 5, targetCy + 5, '#00FF00', -1, false, true)   -- 目标位置
opcvOps.saveMat(vis, saveDir .. 'slider_puzzle_result.png')


-- 12. 释放 Mat 资源
opcvOps.clearMats()
```

### 算法验证

在 389 张腾讯防水墙样本上：

* **识别到 3 个 puzzle 的成功率：95.9%**（373/389）
* 距离范围 0~295 像素（取决于目标缺口位置）
* 距离均值 ~145 像素

### 适用性说明

* **适配分辨率**：360x361 是腾讯防水墙的常见大小，算法对此最优；其他分辨率只要保持宽高比且 puzzle 占图 30%~50% 高度也能工作
* **适配光照**：多种背景（山、海、桥、动物、城市等）上验证
* **失败原因**：源拼图轮廓完全不可见 + 同 Y 行有大尺寸干扰（如屋顶、栏杆），此时会选错 candidate
* **不适用场景**：
  * 拼图形状被大量水印/文字遮挡
  * 3 个拼图不在同一 Y 水平线
  * 拼图尺寸不在 20~100 像素范围

### 与其他滑块识别的对比

| 文档 | 滑块类型 | 核心思路 |
|------|----------|----------|
| K歌滑块（上方） | 1 个音符方块，右侧滑轨区 | 几何特征筛选（宽高比近方形） |
| 本实例 | 3 个拼图横向排列 | 多阈值候选 + 同 Y 行 + 模板匹配 |

### 调试小贴士

1. **检查截图区域**：确保 LuaEngine.snapShot 覆盖整个拼图验证码区域
2. **保存中间结果**：用 `opcvOps.saveMat` 保存源图，送到 PC 上目视检查
3. **调整 Y 范围**：如果你的拼图位置特殊，修改 Lua 中的 `cy >= 60 and cy <= 220`
4. **降低 precision**：`findImg` 的 precision 参数（第 4 个）控制匹配阈值，传 0.0 取最大分位置
5. **多次截图**：某些验证码会动态变化，建议间隔 100~300ms 多次截图取结果一致性最高者
